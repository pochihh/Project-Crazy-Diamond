cmake_minimum_required(VERSION 3.16)
project(bridge)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(yaml_cpp_vendor REQUIRED)
find_package(yaml-cpp REQUIRED)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
set(DIAMOND_PROJECT heart_phantom CACHE STRING "Project controller/configuration directory")
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../../projects/${DIAMOND_PROJECT}/bridge.yaml")
  message(FATAL_ERROR "Unknown DIAMOND_PROJECT: ${DIAMOND_PROJECT}")
endif()
include_directories(include ../../../dsp/host)
add_executable(bridge_node src/bridge_node.cpp)
target_compile_definitions(bridge_node PRIVATE DIAMOND_PROJECT_ID="${DIAMOND_PROJECT}")
ament_target_dependencies(bridge_node rclcpp interfaces sensor_msgs)
target_link_libraries(bridge_node pthread yaml-cpp)
install(TARGETS bridge_node DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(core_test test/core_test.cpp)
  target_compile_options(core_test PRIVATE -UNDEBUG)
  add_test(NAME core_test COMMAND core_test)
  add_executable(protocol_test ../../../tests/protocol_test.cpp)
  target_compile_options(protocol_test PRIVATE -UNDEBUG)
  add_test(NAME protocol_test COMMAND protocol_test)
  add_executable(profiles_test test/profiles_test.cpp)
  target_link_libraries(profiles_test yaml-cpp)
  target_compile_options(profiles_test PRIVATE -UNDEBUG)
  add_test(NAME profiles_test COMMAND profiles_test)
endif()
ament_package()
